# What the packaging environment needs on top of the project's own
# dependencies. Pinned to a floor rather than an exact version: PyInstaller
# tracks new Python releases, and holding it back is how a build breaks on a
# newer interpreter than the one this was written against.
pyinstaller>=6.3

# yt-dlp ships inside the executable, so a packaged build never asks the
# user to install it. Unpinned on purpose: YouTube breaks yt-dlp regularly
# and the fix is always the newest release.
#
# The [default] extra is not optional here: it is what pulls in yt-dlp-ejs,
# the challenge solver scripts the JavaScript runtime actually runs. Without
# them the bundled Deno has nothing to execute and every YouTube video fails
# - with an error that blames the runtime, whichever one is installed.
yt-dlp[default]
